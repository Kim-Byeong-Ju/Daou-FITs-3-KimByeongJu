### 1. 구조체

- **`개념`**
    - **C 언어에서 구조체는 서로 다른 형태의 여러 데이터를 하나로 묶어서 관리할 수 있는 사용자 정의 데이터 형식이다.**
    - **구조체로 만들어진 자료는 연속된 메모리에 할당된다.**
- **`구조체 선언 형식`**
    
    ```c
    struct Student {
        int student_id;
        char name[50];
        int age;
        float gpa;
    };
    ```
    

- **`구조체 사용`**
    
    ```c
    #define _CRT_SECURE_NO_WARNINGS
    #include <stdio.h>
    #include <string.h>
    
    // 방법 1
    struct student {
        char name[12];
        int age;
        char phone[12];
    };
    // 방법 2
    struct student{
        char name[12];
        int age;
        char phone[12];
    } s3, s4; // 전역 변수
    // 방법 3 -- 가장 많이 사용하는 방법
    typedef struct student2 {
        char name[12];
        int age;
        char phone[12];
    
    } ST;
    
    int main() {
        struct student s1, s2; // 지역변수
        s1.age = 10;
        strcpy(s1.name, "홍길동");
        s3.age = 26; 
        ST s5, s6;
        return 0;
    }
    
    ```
    

```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <string.h>

struct student {
    char name[12];
    int id;
    float avg;
};

int main() {
    char name[12];
    int id;
    struct student s1[2];
    for (int i = 0; i < 2; i++) {
        int score;
        int sum = 0;
        printf("학생 %d의 정보 입력: \n", i+1);
        printf("이름: ");
        scanf("%s", name);
        printf("학번: ");
        scanf("%d", &id);
        printf("과목 1 성적: ");
        scanf("%d", &score);
        sum += score;
        printf("과목 2 성적: ");
        scanf("%d", &score);
        sum += score;
        printf("과목 3 성적: ");
        scanf("%d", &score);
        sum += score;
        strcpy(s1[i].name, name);
        s1[i].id = id;
        s1[i].avg = sum / 3.0;
    }
    printf("학생 성적 정보: \n");
    for (int i = 0; i < 2; i++) {
        printf("이름: %s, 학번: %d, 평균: %.2lf\n", s1[i].name, s1[i].id, s1[i].avg);
    }
    return 0;
}

```

```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <string.h>

struct bookVO {
    char title[50];
    char author[20];
    int year;
    char genre[20];
};

int main() {
    struct bookVO book[2];
    char title[50];
    char author[20];
    int year;
    char genre[20];
    for (int i = 0; i < 2; i++) {
        printf("도서 %d의 정보 입력: \n", i + 1);
        printf("제목: ");
        scanf("%s", title);
        printf("저자: ");
        scanf("%s", author);
        printf("출판년도: ");
        scanf("%d", &year);
        printf("장르: ");
        scanf("%s", genre);
        strcpy(book[i].title, title);
        strcpy(book[i].author, author);
        book[i].year = year;
        strcpy(book[i].genre, genre);        
    }
    printf("장르를 입력하세요: ");
    scanf("%s", genre);
    printf("'%s' 장르의 책들: \n", genre);
    for (int i = 0; i < 2; i++) {
        if (strcmp(book[i].genre, genre) == 0) {
            printf("제목: %s, 저자: %s, 출판년도: %d\n", book[i].title, book[i].author, book[i].year);
        }
    }

    return 0;
}

```

### 2. typedef

`typedef`는 **C 언어에서 새로운 데이터 타입을 정의할 때 사용하는 키워드**

기존 타입에 별칭(Alias)을 부여하여 **코드를 더 가독성 있게 만들고, 유지보수를 쉽게** 할 수 있다.

```c
typedef unsigned int uint;

int main() {
    uint a = 100;  // 이제 uint는 unsigned int를 의미함
    printf("%u\n", a);  
    return 0;
}

```
