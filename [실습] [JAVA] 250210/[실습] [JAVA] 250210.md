package first;
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;

public class Practice {
    public static void main(String[] args) {
        Connection con = null;
        PreparedStatement pstmt = null;
        ResultSet rs = null;

        try {
            Class.forName("oracle.jdbc.driver.OracleDriver");
            System.out.println("Driver Loading 성공!");
            String jdbcUrl = "jdbc:oracle:thin:@localhost:1521:xe";
            con = DriverManager.getConnection(jdbcUrl, "C##DEV", "2941");
            System.out.println("데이터베이스 연결 성공");

            String keyword = "여행";

            String sql = "select bisbn, btitle, bauthor, bprice from book where btitle like ?";

            pstmt = con.prepareStatement(sql);

            pstmt.setString(1, "%" + keyword + "%");

            rs = pstmt.executeQuery();

            while(rs.next()) {
                String tmp = "";
                tmp += rs.getString("bisbn") + "  ";
                tmp += rs.getString("btitle") + "  ";
                tmp += rs.getString("bauthor") + "  ";
                tmp += rs.getInt("bprice") + "  ";
                System.out.println(tmp);
            }

        } catch (Exception e) {
            System.out.println(e.getMessage());
        } finally {
            try {
                rs.close();
                pstmt.close();
                if (con != null)
                    con.close();
                System.out.println("connection close()");
            } catch (Exception e2) {
            }
        }
    }
}

// Transaction 관련 코드

package first;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;

// JDBC는 크게 6가지 단계
// 1. (JDBC)Driver Loading: 프로그램에서 딱 1번만 수행
// 2. 데이터베이스 연결: 연결이 성공하면 연결 성공의 증표로 객체가 생성!
//                    객체의 Class는 java.sql.Connection
//                    연결을 하려면 3가지 정보가 필요함
//                    1) 어떤 DB에 접속하는지에 대한 정보(문자열) => JDBC URL(정해진 형식이 있음)
//                    2) ID/PW 필요
// 3. SQL 작성: PreparedStatement 생성
// 4. Query 실행
//    - executeUpdate(): SQL 구문이 insert, update, delete 계열일 때 사용.
//    - executeQuery(): SQL 구문이 select 일 때 사용.
// 5. 결과 처리
//    - executeUpdate()인 경우 리턴값은 SQL에 의해 영향을 받은 행의 수(int)
//    - executeQuery()인 경우 ResultSet이 결과로 나옴
// 6. 리소스 해제
//      - 지금까지 사용된 자원
//      - Connection, PreparedStatement, ResultSet
//      - 사용된 자원을 해지해야 할 때

public class TransactionTest {
    public static void main(String[] args) {
        // 기본적인 JDBC 코드를 생성
        Connection con = null;
        try {
            
            // 1. Driver Loading
            Class.forName("oracle.jdbc.driver.OracleDriver");
            
            // 2. Connection 연결
            String url = "jdbc:oracle:thin:@localhost:1521:xe";
            String user = "C##DEV";
            String password = "2941";
            con = DriverManager.getConnection(url, user, password);
            // 기본적으로 autoCommit mode로 Connection 생성
            // 그러나, 직접 transaction을 설정하려면
            con.setAutoCommit(false); // transaction 시작
            
            // 이 때, 명시적으로 con.commit() 혹은 con.rollback()이 호출되면 transaction 종료
            // 추가적으로 해당 구문이 나오지 않고 connection이 정상 종료되면 con.commit()으로 간주
            
            // 3. SQL 구문 작성
            String sql = "INSERT INTO memberTbl VALUES(?,?)";
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, "1"); // 첫번째 물음표에 대한 값 설정
            pstmt.setString(2, "홍길동"); // 두번째 물음표에 대한 값 설정
            
            // 4. SQL 구문 실행
            int result = pstmt.executeUpdate(); // return 값은 영향 받은 row 수
            // 5. 결과 처리
            if (result == 1) {
                System.out.println("잘 처리됨");
            } else {
                System.out.println("잘 안됨");
            }
            // con.rollback();// 트랜잭션을 무효화하는 코드
            con.commit(); // 트랜잭션 종료
            
            // 6. 리소스 해제
            pstmt.close();
            con.close();
        } catch (Exception e) {
            System.out.println(e.getMessage());
        }
    }
}
