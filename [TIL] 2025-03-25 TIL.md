```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <oci.h>
#include "transaction.h"

void connect_db(DBConnection* db) {
    sword status;
    char* username = "C##DEV";
    char* password = "2941";
    char* dbname = "localhost:1521/xe";
    OCIEnvCreate(&db->envhp, OCI_DEFAULT, NULL, NULL, NULL, NULL, 0, NULL);
    OCIHandleAlloc(db->envhp, (void**)&db->errhp, OCI_HTYPE_ERROR, 0, NULL);
    OCIHandleAlloc(db->envhp, (void**)&db->srvhp, OCI_HTYPE_SERVER, 0, NULL);
    OCIServerAttach(db->srvhp, db->errhp, (OraText*)dbname, strlen(dbname), OCI_DEFAULT);
    OCIHandleAlloc(db->envhp, (void**)&db->svchp, OCI_HTYPE_SVCCTX, 0, NULL);
    OCIAttrSet(db->svchp, OCI_HTYPE_SVCCTX, db->srvhp, 0, OCI_ATTR_SERVER, db->errhp);
    OCIHandleAlloc(db->envhp, (void**)&db->usrhp, OCI_HTYPE_SESSION, 0, NULL);
    OCIAttrSet(db->usrhp, OCI_HTYPE_SESSION, username, strlen(username), OCI_ATTR_USERNAME, db->errhp);
    OCIAttrSet(db->usrhp, OCI_HTYPE_SESSION, password, strlen(password), OCI_ATTR_PASSWORD, db->errhp);
    OCISessionBegin(db->svchp, db->errhp, db->usrhp, OCI_CRED_RDBMS, OCI_DEFAULT);
    OCIAttrSet(db->svchp, OCI_HTYPE_SVCCTX, db->usrhp, 0, OCI_ATTR_SESSION, db->errhp);
    db->stmthp = NULL;
    printf("Oracle DB 연결 성공!\n");
}

void disconnect_db(DBConnection* db) {
    if (db->stmthp) OCIHandleFree(db->stmthp, OCI_HTYPE_STMT);
    if (db->svchp && db->errhp) OCILogoff(db->svchp, db->errhp);
    if (db->usrhp) OCIHandleFree(db->usrhp, OCI_HTYPE_SESSION);
    if (db->svchp) OCIHandleFree(db->svchp, OCI_HTYPE_SVCCTX);
    if (db->srvhp) OCIHandleFree(db->srvhp, OCI_HTYPE_SERVER);
    if (db->errhp) OCIHandleFree(db->errhp, OCI_HTYPE_ERROR);
    if (db->envhp) OCIHandleFree(db->envhp, OCI_HTYPE_ENV);
    printf("연결 종료\n");
}

void add_transaction(DBConnection* db, int transaction_id, char* customer_name, char* stock_name, char transaction_type, int quantity, double price) {
    OCIStmt* stmthp;
    sword status;
    char* sql = "INSERT INTO stock_transaction (transaction_id, customer_name, stock_name, transaction_type, quantity, price) "
        "VALUES (:1, :2, :3, :4, :5, :6)";
    // 스테이트먼트 핸들 할당
    OCIHandleAlloc(db->envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);
    // SQL 문 준비
    OCIStmtPrepare(stmthp, db->errhp, (OraText*)sql, (ub4)strlen(sql), OCI_NTV_SYNTAX, OCI_DEFAULT);
    // 바인드 변수 설정
    OCIBind* bind1 = NULL;
    OCIBind* bind2 = NULL;
    OCIBind* bind3 = NULL;
    OCIBind* bind4 = NULL;
    OCIBind* bind5 = NULL;
    OCIBind* bind6 = NULL;

    OCIBindByName(stmthp, &bind1, db->errhp, (OraText*)":1", -1, &transaction_id, sizeof(transaction_id), SQLT_INT, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
    OCIBindByName(stmthp, &bind2, db->errhp, (OraText*)":2", -1, customer_name, strlen(customer_name) + 1, SQLT_STR, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
    OCIBindByName(stmthp, &bind3, db->errhp, (OraText*)":3", -1, stock_name, strlen(stock_name) + 1, SQLT_STR, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
    OCIBindByName(stmthp, &bind4, db->errhp, (OraText*)":4", -1, &transaction_type, sizeof(transaction_type), SQLT_AFC, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
    OCIBindByName(stmthp, &bind5, db->errhp, (OraText*)":5", -1, &quantity, sizeof(quantity), SQLT_INT, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
    OCIBindByName(stmthp, &bind6, db->errhp, (OraText*)":6", -1, &price, sizeof(price), SQLT_FLT, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
    // SQL 실행
    status = OCIStmtExecute(db->svchp, stmthp, db->errhp, 1, 0, NULL, NULL, OCI_COMMIT_ON_SUCCESS);

    if (status != OCI_SUCCESS) {
        printf("트랜잭션 추가 실패\n");
    }
    else {
        printf("트랜잭션 추가 성공\n");
    }

    // 스테이트먼트 핸들 해제
    OCIHandleFree(stmthp, OCI_HTYPE_STMT);

}

void display_transactions(DBConnection* db) {
    OCIStmt* stmthp;
    sword status;
    char* sql = "SELECT transaction_id, customer_name, stock_name, transaction_type, quantity, price FROM stock_transaction";

    // 스테이트먼트 핸들 할당
    OCIHandleAlloc(db->envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);

    // SQL 문 준비
    OCIStmtPrepare(stmthp, db->errhp, (OraText*)sql, (ub4)strlen(sql), OCI_NTV_SYNTAX, OCI_DEFAULT);

    // SQL 실행
    status = OCIStmtExecute(db->svchp, stmthp, db->errhp, 0, 0, NULL, NULL, OCI_DEFAULT);
    if (status != OCI_SUCCESS) {
        printf("거래 내역 조회 실패\n");
        OCIHandleFree(stmthp, OCI_HTYPE_STMT);
        return;
    }

    // 결과 저장할 변수들
    int transaction_id;
    char customer_name[50];
    char stock_name[50];
    char transaction_type;
    int quantity;
    double price;

    // 결과 매핑 (OCIDefineByPos)
    OCIDefine* def1 = NULL;
    OCIDefine* def2 = NULL;
    OCIDefine* def3 = NULL;
    OCIDefine* def4 = NULL;
    OCIDefine* def5 = NULL;
    OCIDefine* def6 = NULL;

    OCIDefineByPos(stmthp, &def1, db->errhp, 1, &transaction_id, sizeof(transaction_id), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def2, db->errhp, 2, customer_name, sizeof(customer_name), SQLT_STR, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def3, db->errhp, 3, stock_name, sizeof(stock_name), SQLT_STR, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def4, db->errhp, 4, &transaction_type, sizeof(transaction_type), SQLT_AFC, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def5, db->errhp, 5, &quantity, sizeof(quantity), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def6, db->errhp, 6, &price, sizeof(price), SQLT_FLT, NULL, NULL, NULL, OCI_DEFAULT);

    // 결과 출력
    printf("-----------------------------------------------------\n");
    printf(" ID | Customer   | Stock    | Type | Quantity | Price \n");
    printf("-----------------------------------------------------\n");

    while ((status = OCIStmtFetch2(stmthp, db->errhp, 1, OCI_DEFAULT, 0, OCI_DEFAULT)) == OCI_SUCCESS || status == OCI_SUCCESS_WITH_INFO) {
        printf("%3d | %-10s | %-8s | %4c | %8d | %10.2f\n",
            transaction_id, customer_name, stock_name, transaction_type, quantity, price);
    }

    printf("-----------------------------------------------------\n");

    // 스테이트먼트 핸들 해제
    OCIHandleFree(stmthp, OCI_HTYPE_STMT);
}

void delete_transaction(DBConnection* db, int transaction_id) {
    OCIStmt* stmthp;
    sword status;
    char* sql = "DELETE FROM stock_transaction WHERE transaction_id = :1";

    // 스테이트먼트 핸들 할당
    OCIHandleAlloc(db->envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);

    // SQL 문 준비
    OCIStmtPrepare(stmthp, db->errhp, (OraText*)sql, (ub4)strlen(sql), OCI_NTV_SYNTAX, OCI_DEFAULT);

    // 바인드 변수 설정 (transaction_id)
    OCIBind* bnd1 = NULL;
    OCIBindByPos(stmthp, &bnd1, db->errhp, 1, &transaction_id, sizeof(transaction_id), SQLT_INT, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);

    // SQL 실행
    status = OCIStmtExecute(db->svchp, stmthp, db->errhp, 1, 0, NULL, NULL, OCI_DEFAULT);

    if (status != OCI_SUCCESS) {
        printf("거래 삭제 실패 (ID: %d)\n", transaction_id);

    else {
        printf("거래 삭제 성공 (ID: %d)\n", transaction_id);
    }

    // 트랜잭션 커밋
    OCITransCommit(db->svchp, db->errhp, OCI_DEFAULT);

    // 스테이트먼트 핸들 해제
    OCIHandleFree(stmthp, OCI_HTYPE_STMT);
}
void save_to_file(DBConnection* db) {
    OCIStmt* stmthp;
    sword status;
    char* sql = "SELECT transaction_id, customer_name, stock_name, transaction_type, quantity, price FROM stock_transaction";

    FILE* datFile = fopen("transactions.dat", "wb");
    FILE* csvFile = fopen("transactions.csv", "w");
    if (!datFile || !csvFile) {
        printf("파일을 열 수 없습니다.\n");
        return;
    }

    // CSV 헤더 작성
    fprintf(csvFile, "ID,Customer,Stock,Type,Quantity,Price\n");

    OCIHandleAlloc(db->envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);
    OCIStmtPrepare(stmthp, db->errhp, (OraText*)sql, (ub4)strlen(sql), OCI_NTV_SYNTAX, OCI_DEFAULT);
    status = OCIStmtExecute(db->svchp, stmthp, db->errhp, 0, 0, NULL, NULL, OCI_DEFAULT);
    if (status != OCI_SUCCESS) {
        printf("거래 내역 조회 실패\n");
        OCIHandleFree(stmthp, OCI_HTYPE_STMT);
        return;
    }

    int transaction_id;
    char customer_name[50];
    char stock_name[50];
    char transaction_type;
    int quantity;
    double price;

    OCIDefine* def1 = NULL;
    OCIDefine* def2 = NULL;
    OCIDefine* def3 = NULL;
    OCIDefine* def4 = NULL;
    OCIDefine* def5 = NULL;
    OCIDefine* def6 = NULL;

    OCIDefineByPos(stmthp, &def1, db->errhp, 1, &transaction_id, sizeof(transaction_id), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def2, db->errhp, 2, customer_name, sizeof(customer_name), SQLT_STR, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def3, db->errhp, 3, stock_name, sizeof(stock_name), SQLT_STR, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def4, db->errhp, 4, &transaction_type, sizeof(transaction_type), SQLT_AFC, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def5, db->errhp, 5, &quantity, sizeof(quantity), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def6, db->errhp, 6, &price, sizeof(price), SQLT_FLT, NULL, NULL, NULL, OCI_DEFAULT);

    while ((status = OCIStmtFetch2(stmthp, db->errhp, 1, OCI_DEFAULT, 0, OCI_DEFAULT)) == OCI_SUCCESS || status == OCI_SUCCESS_WITH_INFO) {
        fwrite(&transaction_id, sizeof(transaction_id), 1, datFile);
        fwrite(customer_name, sizeof(customer_name), 1, datFile);
        fwrite(stock_name, sizeof(stock_name), 1, datFile);
        fwrite(&transaction_type, sizeof(transaction_type), 1, datFile);
        fwrite(&quantity, sizeof(quantity), 1, datFile);
        fwrite(&price, sizeof(price), 1, datFile);

        fprintf(csvFile, "%d,%s,%s,%c,%d,%.2f\n", transaction_id, customer_name, stock_name, transaction_type, quantity, price);
    }

    fclose(datFile);
    fclose(csvFile);
    printf("거래 내역이 파일로 저장되었습니다.\n");

    OCIHandleFree(stmthp, OCI_HTYPE_STMT);
}
```
