package practice.controller;

import javafx.collections.ObservableList;
import javafx.fxml.FXML;
import javafx.fxml.Initializable;
import javafx.scene.control.*;
import javafx.scene.control.cell.PropertyValueFactory;
import practice.service.BookSearchService;
import practice.service.BookSearchServiceOracleImpl;
import practice.vo.BookVO;

import java.net.URL;
import java.util.ResourceBundle;

public class BookSearchController implements Initializable {

    @FXML
    private TextField textField;
    @FXML
    private Button searchBtn;
    @FXML
    private Button deleteBtn;
    @FXML
    private TableView<BookVO> tableView;
    @FXML
    private TableColumn<BookVO, String> titleCol;
    @FXML
    private TableColumn<BookVO, String> authorCol;
    @FXML
    private TableColumn<BookVO, Integer> priceCol;
    @FXML
    private TableColumn<BookVO, String> isbnCol;

    private String deleteIsbn;
    private final BookSearchService service = new BookSearchServiceOracleImpl();

    @Override
    public void initialize(URL url, ResourceBundle resourceBundle) {
        isbnCol.setCellValueFactory(new PropertyValueFactory<>("bisbn"));
        titleCol.setCellValueFactory(new PropertyValueFactory<>("btitle"));
        authorCol.setCellValueFactory(new PropertyValueFactory<>("bauthor"));
        priceCol.setCellValueFactory(new PropertyValueFactory<>("bprice"));

        tableView.setRowFactory(e -> {
            TableRow<BookVO> row = new TableRow<>();
            row.setOnMouseClicked(event -> {
                if (event.getClickCount() == 1) {
                    BookVO book = row.getItem();
                    if (book != null) {
                        deleteIsbn = book.getBisbn();
                    }
                }
            });
            return row;
        });

        searchBtn.setOnAction(event -> searchTableView());
        deleteBtn.setOnAction(event -> deleteBook());
    }

    private void searchTableView() {
        try {
            ObservableList<BookVO> list = service.searchBook(textField.getText());
            tableView.setItems(list);
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }

    private void deleteBook() {
        if (deleteIsbn != null) {
            try {
                service.deleteBook(deleteIsbn, textField.getText());
                searchTableView();
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        }
    }
}

package practice.dao;

import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import practice.vo.BookVO;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;

public class BookDAO {

    private Connection con;
    public BookDAO(Connection con) {
        this.con = con;
    }

    public ObservableList<BookVO> select(String text) {
        ObservableList<BookVO> books = FXCollections.observableArrayList();
        try {
            String sql = "select bisbn, btitle, bprice, bauthor from book where btitle like ?";
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, "%" + text + "%");
            ResultSet rs = pstmt.executeQuery();
            while (rs.next()) {
                BookVO book = new BookVO();
                book.setBisbn(rs.getString("bisbn"));
                book.setBtitle(rs.getString("btitle"));
                book.setBauthor(rs.getString("bauthor"));
                book.setBprice(rs.getInt("bprice"));
                books.add(book);
            }
            rs.close();
            pstmt.close();
        } catch (Exception e) {
            e.printStackTrace();
        }

        return books;
    }

    public void update(String isbn, String field, String value)  {
        String sql = "update book set ? = ? where bisbn = ?";
        try {
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, field);
            pstmt.setString(2, value);
            pstmt.setString(3, isbn);
            pstmt.executeUpdate();
            pstmt.close();
        } catch (Exception e) {
        e.printStackTrace();
        }
    }

    public void delete(String isbn) {
        try {
            String sql = "delete from book where bisbn = ?";
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, isbn);
            pstmt.executeUpdate();
            pstmt.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
