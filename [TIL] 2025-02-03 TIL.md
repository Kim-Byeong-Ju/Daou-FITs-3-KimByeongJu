# TIL: 프로젝트 발표

## 학습 날짜: 2025년 2월 3일

## 오늘 배운 내용 정리

- 커밋 후 롤백

  ```sql
  FLASHBACK TABLE ACCOUNTS TO TIMESTAMP (SYSTIMESTAMP - INTERVAL '10' MINUTE);
  ALTER TABLE ACCOUNTS ENABLE ROW MOVEMENT;
  ```

- 프로젝트 db 스키마 쿼리문
  - ORACLE에서는 TEXT 데이터 타입이 존재하지 않는다. 긴 문자열은 LONG, CLOB 자료형을 지정해야 함.
  - LONG: Table에 한 개의 컬럼만 생성할 수 있다. 2GB의 문자열 데이터를 저장할 수 있다.
  - CLOB: Table에 여러개의 컬럼을 생성할 수 있다. 4GB의 문자열 데이터를 저장할 수 있다. 

  ```sql
    CREATE TABLE reports (
      report_date VARCHAR(8) NOT NULL,
      stock_id VARCHAR(64) NOT NULL,
      previous_close NUMBER(8),
      open NUMBER(8),
      high NUMBER(8),
      low NUMBER(8),
      volume NUMBER(10),
      creation_timestamp TIMESTAMP DEFAULT SYSTIMESTAMP,
      update_timestamp TIMESTAMP DEFAULT SYSTIMESTAMP,
      CONSTRAINT reports_pk PRIMARY KEY (report_date, stock_id)
      );
  
  CREATE TABLE users (
      user_id VARCHAR(64) NOT NULL,
      password VARCHAR(64),
      gender VARCHAR(4),
      name VARCHAR(20),
      phone_number VARCHAR(20),
      email VARCHAR(40),
      age NUMBER,
      birth_date VARCHAR(8),
      income NUMBER,
      investment_type VARCHAR(4),
      join_date VARCHAR(8),
      withdraw_date VARCHAR(8),
      created_at TIMESTAMP DEFAULT SYSTIMESTAMP,
      updated_at TIMESTAMP DEFAULT SYSTIMESTAMP,
      CONSTRAINT users_pk PRIMARY KEY (user_id)
      );
  
  CREATE TABLE views (
      view_id VARCHAR(64) NOT NULL,
      stock_id VARCHAR(64) NOT NULL,
      user_id VARCHAR(64) NOT NULL,
      view_timestamp TIMESTAMP DEFAULT SYSTIMESTAMP,
      CONSTRAINT views_pk PRIMARY KEY (view_id)
      );
  
  CREATE TABLE TRANSACTIONS (
      transaction_id VARCHAR(64) NOT NULL,
      transaction_date DATE DEFAULT SYSDATE NOT NULL,
      order_id VARCHAR(64) NOT NULL,
      order_date DATE NOT NULL,
      quantity NUMBER,
      price NUMBER,
      change NUMBER,
      created_at TIMESTAMP DEFAULT SYSTIMESTAMP,
      updated_at TIMESTAMP DEFAULT SYSTIMESTAMP,
      CONSTRAINT transactions_pk PRIMARY KEY (transaction_id)
      );
  
  CREATE TABLE accounts (
      account_id VARCHAR(64) NOT NULL,
      user_id VARCHAR(64) NOT NULL,
      account_number VARCHAR(64) UNIQUE,
      status VARCHAR(4),
      type VARCHAR(4),
      password VARCHAR(64),
      open_date VARCHAR(8),
      close_date VARCHAR(8),
      deposit NUMBER,
      withholding NUMBER,
      created_at TIMESTAMP DEFAULT SYSTIMESTAMP,
      updated_at TIMESTAMP DEFAULT SYSTIMESTAMP,
      CONSTRAINT accounts_pk PRIMARY KEY (account_id)
      );
  
  CREATE TABLE stocks (
      stock_id VARCHAR(64) NOT NULL,
      name VARCHAR(20),
      status VARCHAR(4),
      exchange_code VARCHAR(4),
      description LONG,
      sector VARCHAR(4),
      list_date VARCHAR(8),
      delist_date VARCHAR(8),
      created_at TIMESTAMP DEFAULT SYSTIMESTAMP,
      updated_at TIMESTAMP DEFAULT SYSTIMESTAMP,
      CONSTRAINT stocks_pk PRIMARY KEY (stock_id)
      );
  
  CREATE TABLE ORDERS (
      order_id VARCHAR(64) NOT NULL,
      order_date DATE DEFAULT SYSDATE NOT NULL,
      stock_id VARCHAR(64) NOT NULL,
      account_id VARCHAR(64) NOT NULL,
      status VARCHAR(10),
      type VARCHAR(4),
      quantity NUMBER,
      price NUMBER,
      traded NUMBER,
      not_traded NUMBER,
      created_at TIMESTAMP DEFAULT SYSTIMESTAMP,
      updated_at TIMESTAMP DEFAULT SYSTIMESTAMP,
      CONSTRAINT orders_pk PRIMARY KEY (order_id)
      );
  
  CREATE TABLE financials (
      financial_date VARCHAR(8) NOT NULL,
      stock_id VARCHAR(64) NOT NULL,
      quarter NUMBER(1),
      earnings NUMBER(10),
      operating_expenses NUMBER(10),
      net_profit NUMBER(10),
      ebitda NUMBER(10),
      create_timestamp TIMESTAMP DEFAULT SYSTIMESTAMP,
      update_timestamp TIMESTAMP DEFAULT SYSTIMESTAMP,
      CONSTRAINT financials_pk PRIMARY KEY (financial_date, stock_id)
      );
  
  CREATE TABLE holdings (
      stock_id VARCHAR(64) NOT NULL,
      account_id VARCHAR(64) NOT NULL,
      quantity NUMBER,
      total_price NUMBER,
      created_at TIMESTAMP DEFAULT SYSTIMESTAMP,
      updated_at TIMESTAMP DEFAULT SYSTIMESTAMP,
      CONSTRAINT holdings_pk PRIMARY KEY (stock_id, account_id)
      );
```
