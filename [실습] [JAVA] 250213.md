package example.view;

import javafx.application.Application;
import javafx.fxml.FXMLLoader;
import javafx.scene.Parent;
import javafx.scene.Scene;
import javafx.stage.Stage;

public class BookView extends Application {
    @Override
    public void start(Stage stage) throws Exception {
        // 화면 구성
        // Stage - Scene - Parent(Layout Manager)
        Parent root = null;
        FXMLLoader loader = new FXMLLoader(getClass().getResource("/booksearchmybatis.fxml"));
        try {
            root = loader.load();
        } catch (Exception e) {
            e.printStackTrace();
        }
        Scene scene = new Scene(root);
        stage.setScene(scene);
        stage.setTitle("도서 등록/검색");
        stage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}

// controller
package example.controller;

import example.service.BookService;
import example.service.BookServiceImpl;
import example.vo.BookVO;
import javafx.collections.ObservableList;
import javafx.fxml.Initializable;
import javafx.fxml.FXML;
import javafx.scene.control.*;
import javafx.scene.control.cell.PropertyValueFactory;
import javafx.scene.control.cell.TextFieldTableCell;
import javafx.scene.input.KeyCode;
import javafx.util.converter.IntegerStringConverter;

import java.net.URL;
import java.util.ResourceBundle;

public class BookController implements Initializable {

    @FXML
    private TextField searchField;
    @FXML
    private TextField isbnField;
    @FXML
    private TextField titleField;
    @FXML
    private TextField priceField;
    @FXML
    private TextField authorField;
    @FXML
    private Button insertBtn;
    @FXML
    private Button searchBtn;
    @FXML
    private Button deleteBtn;
    @FXML
    private TableView<BookVO> tableView;
    @FXML
    private TableColumn<BookVO, String> titleCol;
    @FXML
    private TableColumn<BookVO, String> authorCol;
    @FXML
    private TableColumn<BookVO, Integer> priceCol;
    @FXML
    private TableColumn<BookVO, String> isbnCol;

    private String deleteIsbn;
    private final BookService service = new BookServiceImpl();
    @Override
    public void initialize(URL url, ResourceBundle resourceBundle) {
        isbnCol.setCellValueFactory(new PropertyValueFactory<>("bisbn"));
        titleCol.setCellValueFactory(new PropertyValueFactory<>("btitle"));
        authorCol.setCellValueFactory(new PropertyValueFactory<>("bauthor"));
        priceCol.setCellValueFactory(new PropertyValueFactory<>("bprice"));

        // 데이터 수정
        tableView.setEditable(true);
        titleCol.setCellFactory(TextFieldTableCell.forTableColumn());
        titleCol.setOnEditCommit(event -> {
            BookVO book = event.getRowValue();
            String newTitle = event.getNewValue();
            book.setBtitle(newTitle);
            updateBook(book);
        });
        priceCol.setCellFactory(TextFieldTableCell.forTableColumn(new IntegerStringConverter()));
        priceCol.setOnEditCommit(event -> {
            BookVO book = event.getRowValue();
            Integer newPrice = event.getNewValue();
            book.setBprice(newPrice);
            updateBook(book);
        });
        authorCol.setCellFactory(TextFieldTableCell.forTableColumn());
        authorCol.setOnEditCommit(event -> {
            BookVO book = event.getRowValue();
            String newAuthor = event.getNewValue();
            book.setBauthor(newAuthor);
            updateBook(book);
        });

        // 데이터 삭제
        tableView.setRowFactory(e -> {
            TableRow<BookVO> row = new TableRow<>();
            row.setOnMouseClicked(event -> {
                if (event.getClickCount() == 1) {
                    BookVO book = row.getItem();
                    if (book != null) {
                        deleteIsbn = book.getBisbn();
                    }
                }
            });
            return row;
        });

        searchBtn.setOnAction(event -> searchTableView());
        deleteBtn.setOnAction(event -> deleteBook());
        insertBtn.setOnAction(event -> insertBook());
    }

    private void searchTableView() {
        try {
            ObservableList<BookVO> list = service.searchBook(searchField.getText());
            tableView.setItems(list);
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }

    private void deleteBook() {
        if (deleteIsbn != null) {
            try {
                service.deleteBook(deleteIsbn);
                searchTableView();
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        }
    }
    private void insertBook() {
        BookVO book = new BookVO();
        book.setBisbn(isbnField.getText());
        book.setBtitle(titleField.getText());
        book.setBprice(Integer.parseInt(priceField.getText()));
        book.setBauthor(authorField.getText());
        try {
            service.insertBook(book);
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }
    private void updateBook(BookVO book) {
        try {
            service.updateBook(book);
            searchTableView();
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }
}

// service

package example.service;

import example.dao.BookDAO;
import example.mybatis.MyBatisSessionFactory;
import example.vo.BookVO;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;

import java.awt.print.Book;
import java.util.List;

public class BookServiceImpl implements BookService {

    @Override
    public ObservableList<BookVO> searchBook(String keyword) throws Exception {
        SqlSessionFactory factory = MyBatisSessionFactory.getSqlSessionFactory();
        SqlSession sqlSession = factory.openSession();
        ObservableList<BookVO> books = null;
        try {
            BookDAO dao = new BookDAO(sqlSession);
            List<BookVO> bookList  = dao.selectByKeyword(keyword);
            books = FXCollections.observableArrayList(bookList);
        } catch (Exception e) {
            throw new Exception("search 트랜잭션 처리 중 오류 발생", e);
        } finally {
            sqlSession.close();
        }
        return books;
    }

    @Override
    public void deleteBook(String isbn) throws Exception {
        SqlSessionFactory factory = MyBatisSessionFactory.getSqlSessionFactory();
        SqlSession sqlSession = factory.openSession();
        try {
            BookDAO dao = new BookDAO(sqlSession);
            dao.delete(isbn);
            sqlSession.commit();
        } catch (Exception e) {
            sqlSession.rollback();
            throw new Exception("delete 트랜잭션 처리 중 오류 발생", e);
        } finally {
            sqlSession.close();
        }
    }

    @Override
    public void insertBook(BookVO bookVO) throws Exception {
        SqlSessionFactory factory = MyBatisSessionFactory.getSqlSessionFactory();
        SqlSession sqlSession = factory.openSession();
        try {
            BookDAO dao = new BookDAO(sqlSession);
            dao.insert(bookVO);
            sqlSession.commit();
        } catch (Exception e) {
            sqlSession.rollback();
            throw new Exception("insert 트랜잭션 처리 중 오류 발생", e);
        } finally {
            sqlSession.close();
        }

    }

    @Override
    public void updateBook(BookVO bookVO) throws Exception {
        SqlSessionFactory factory = MyBatisSessionFactory.getSqlSessionFactory();
        SqlSession sqlSession = factory.openSession();
        try {
            BookDAO dao = new BookDAO(sqlSession);
            dao.update(bookVO);
            sqlSession.commit();
        } catch (Exception e) {
            sqlSession.rollback();
            throw new Exception("update 트랜잭션 처리 중 오류 발생", e);
        } finally {
            sqlSession.close();
        }
    }
}

package example.dao;

// sqlSessionFactory를 DAO에게 전달
import example.vo.BookVO;
import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;
import java.util.List;

public class BookDAO {

    private SqlSession SqlSession;

    public BookDAO(SqlSession SqlSession) {
        this.SqlSession = SqlSession;
    }
    // 키워드 검색
    public List<BookVO> selectByKeyword(String keyword) {
        List<BookVO> list = null;
        try {
            list = SqlSession.selectList("example.MyBook.selectByKeyword", keyword);
        } catch (Exception e) {
            e.printStackTrace();
        }
        return list;
    }
    // 삭제
    public void delete(String bisbn) {
        try {
            SqlSession.delete("example.MyBook.delete", bisbn);
            System.out.println("삭제 완료");
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
    // 삽입
    public void insert(BookVO bookVO) {
        try {
            SqlSession.insert("example.MyBook.insert", bookVO);
            System.out.println("정상 등록");
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
    // 수정
    public void update(BookVO bookVO) {
        try {
            SqlSession.update("example.MyBook.update", bookVO);
            System.out.println("수정 완료");
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
