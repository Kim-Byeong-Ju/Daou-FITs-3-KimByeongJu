package javaexam2;
import java.util.*;
public class Test03 {
    public static void main(String[] args) {
        String str = "1.22,4.12,5.93,8.71,9.34";
        double data [] = new double[5];
        double sum = 0;
        StringTokenizer st = new StringTokenizer(str, ",");
        for(int i=0; st.hasMoreElements(); i++) {
            data[i] = Double.parseDouble(st.nextToken());
        }

        for(double d : data) {
            sum += d;
        }
        System.out.printf("합계: %.3f%n", sum);
        System.out.printf("평균: %.3f%n", sum / data.length);
    }
}

package javaexam2;

    public static void main(String[] args) {
        HashMap<Integer, Integer> map = new HashMap<Integer, Integer>();
        double sum = 0;
        Random rand = new Random();
        for (int i = 1; i < 11; i++) {
            int num = rand.nextInt(100)+1;
            map.put(i, num);
        }
        for (Map.Entry<Integer, Integer> entry : map.entrySet()) {
            System.out.print(entry.getValue()+" ");
            sum += entry.getValue();
        }
        System.out.println();
        System.out.printf("합계: %.2f%n", sum);
        System.out.printf("평균: %.2f%n", sum/map.size());
    }
}

package javaexam2.plane;

public abstract class Plane {
    private String planeName;
    private int fuelSize;

    public Plane() {
    }
    public Plane(String planeName, int fuelSize) {
        this.planeName = planeName;
        this.fuelSize = fuelSize;
    }
    public void refuel(int fuel) {
        setFuelSize(this.getFuelSize() + fuel);
    };
    public abstract void flight(int distance);

    public String getPlaneName() {
        return planeName;
    }

    public void setPlaneName(String planeName) {
        this.planeName = planeName;
    }

    public int getFuelSize() {
        return fuelSize;
    }

    public void setFuelSize(int fuelSize) {
        this.fuelSize = fuelSize;
    }
}

package javaexam2.plane;

public class Cargoplane extends Plane{
    public Cargoplane() {
    }
    public Cargoplane(String planeName, int fuelSize) {
        super(planeName, fuelSize);
    }
    @Override
    public void flight(int distance){
        setFuelSize(getFuelSize() - distance*5);
    }
}

package javaexam2.plane;

public class Airplane extends Plane{
    public Airplane(){
    }
    public Airplane(String planeName, int fuelSize){
        super(planeName, fuelSize);
    }
    @Override
    public void flight(int distance){
        setFuelSize(getFuelSize() - distance*3);
    }
}

package javaexam2.plane;

public class PlaneTest {
    public static void main(String[] args) {
        // 객체 생성
        Airplane airplane = new Airplane("L747", 1000);

        System.out.println("   "+cargoplane.getPlaneName()+"       "+cargoplane.getFuelSize());

package javaexam2.company;

public abstract class Employee {
    private String name;
    private int number;
    private String department;
    private int salary;

    public Employee() {
    }
    public Employee(String name, int number, String department, int salary) {
        this.name = name;
        this.number = number;
        this.department = department;
        this.salary = salary;
    }
    public abstract double tax();
    @Override
    public String toString() {
        return name + "      " + number + "     " + department;
    }
    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public int getNumber() {
        return number;
    }

    public void setNumber(int number) {
        this.number = number;
    }

    public String getDepartment() {
        return department;
    }

    public void setDepartment(String department) {
        this.department = department;
    }

    public int getSalary() {
        return salary;
    }

    public void setSalary(int salary) {
        this.salary = salary;
    }
}

package javaexam2.company;

public interface Bonus {
    public void incentive(int pay);
}

package javaexam2.company;

public class Secretary extends Employee implements Bonus{
    public Secretary() {
    }
    public Secretary(String name, int number, String department, int salary) {
        super(name, number, department, salary);
    }
    @Override
    public double tax() {
        return getSalary() * 0.1;
    }
    @Override
    public void incentive(int pay) {
        setSalary(getSalary() + (int) (pay*0.8));
    }
}

package javaexam2.company;

public class Sales extends Employee implements Bonus{
    public Sales() {
    }
    public Sales(String name, int number, String department, int salary) {
        super(name, number, department, salary);
    }
    @Override
    public double tax() {
        return super.getSalary() * 0.13;
    }
    @Override
    public void incentive(int pay) {
        setSalary(getSalary() + (int) (pay*1.2));
    }
}

package javaexam2.company;

import java.util.HashMap;

public class Company {
    public static void main(String[] args) {
        HashMap<Integer, Employee> map = new HashMap<Integer, Employee>();

        map.put(s2.getNumber(), s2);

        System.out.println("------------------------------");

            System.out.println(value.toString());
        }

        for (Employee value : map.values()) {
            System.out.println(value.getName() + "    " + value.getDepartment() + "    " + value.getSalary() + "       " + value.tax());
        }
    }
}
